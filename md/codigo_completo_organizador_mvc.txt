    // MÃ©todo estÃ¡tico para checar se o usuÃ¡rio estÃ¡ logado e sessÃ£o vÃ¡lida
    
    public static function checkLogin() {
        if (!isset($_SESSION['usuario_logado'])) {
            return false;
        }

        if (isset($_SESSION['LAST_ACTIVITY']) && (time() - $_SESSION['LAST_ACTIVITY'] > self::SESSION_TIMEOUT)) {
            session_unset();
            session_destroy();
            return false;
        }

        $_SESSION['LAST_ACTIVITY'] = time();
        return true;
    }

    // Exibe a pÃ¡gina de login
    public function login() {
        // Inicializa variÃ¡veis que serÃ£o enviadas para a view
        $sessao_expirada = false;
        $erro_login = '';

        if (isset($_GET['expirado']) && $_GET['expirado'] == 1) {
            $sessao_expirada = true;
        }

        // Busca mensagem de erro de login na sessÃ£o, se existir
        if (isset($_SESSION['erro_login'])) {
            $erro_login = $_SESSION['erro_login'];
            unset($_SESSION['erro_login']); // Limpa o erro para prÃ³xima requisiÃ§Ã£o
        }

        // Retorna os dados que serÃ£o usados na view
        return [
            'sessao_expirada' => $sessao_expirada,
            'erro_login' => $erro_login
        ];
    }

    // Processa o formulÃ¡rio de login enviado pelo usuÃ¡rio
    public function loginSubmit() {
        $login = trim($_POST['login'] ?? ''); // Remove espaÃ§os extras
        $senha = $_POST['senha'] ?? ''; // Captura senha

        // ValidaÃ§Ã£o bÃ¡sica: campos obrigatÃ³rios
        if (empty($login) || empty($senha)) {
            $_SESSION['erro_login'] = "Login e senha sÃ£o obrigatÃ³rios.";
            header("Location: index.php?controller=auth&method=login");
            exit;
        }

         // Busca usuÃ¡rio no banco via model
        $usuario = $this->authModel->findByLogin($login);

        // Valida se usuÃ¡rio existe e senha estÃ¡ correta
        if ($usuario && password_verify($senha, $usuario['usu_senha'])) {
            // Cria sessÃ£o para usuÃ¡rio logado
            $_SESSION['usuario_logado'] = $login;
            $_SESSION['usuario_nome'] = $usuario['usu_nome'];
            $_SESSION['usuario_id'] = $usuario['usu_id'];
            $_SESSION['LAST_ACTIVITY'] = time();

            // Flag para mostrar splash apenas apÃ³s login
            $_SESSION['show_splash'] = true;

            // Redireciona para pÃ¡gina inicial apÃ³s login
            header("Location: index.php?controller=home&method=index");
            exit;
        } else {
            // UsuÃ¡rio nÃ£o encontrado ou senha incorreta
            $_SESSION['erro_login'] = $usuario ? "Senha incorreta." : "UsuÃ¡rio nÃ£o encontrado.";
            header("Location: index.php?controller=auth&method=login");
            exit;
        }
    }

            if ($erroPrincipal || $erroCheckbox) {

                ]
            );
        } catch (PDOException $e) {
            die("Erro de conexÃ£o: " . $e->getMessage());             // Finaliza a execuÃ§Ã£o se nÃ£o conectar

    // Insere nova tarefa no banco com os dados recebidos do controller
    public function criarTarefa(array $dados): array {
        try {
            $stmt = $this->pdo->prepare("
                INSERT INTO TAREFAS (
                    TASK_TITULO,
                    TASK_DESCRICAO,
                    TASK_TIPOTAREFA,
                    TASK_PRIORIDADE,
                    TASK_DATAFIM,
                    USU_ID
                ) VALUES (
                    :titulo,
                    :descricao,
                    :tipotarefa,
                    :prioridade,
                    :datafim,
                    :usu_id
                )
            ");

            // Executa a query com os dados passados
            $stmt->execute([
                ':titulo'      => $dados['titulo'],
                ':descricao'   => $dados['descricao'] ?? null,
                ':tipotarefa'  => $dados['tipotarefa'],
                ':prioridade'  => $dados['prioridade'] ?? 'mÃ©dia', // padrÃ£o em minÃºsculo para casar com a view
                ':datafim'     => $dados['datafim'] ?? null,
                ':usu_id'      => $dados['usu_id']
            ]);

            // Retorna sucesso com o ID da tarefa criada
            return [
                'sucesso' => true,
                'id' => $this->pdo->lastInsertId()
            ];
        } catch (PDOException $e) {
            // Em caso de erro retorna array com sucesso=false e mensagem de erro
            return [
                'sucesso' => false,
                'erro' => "Erro ao criar tarefa: " . $e->getMessage()
            ];
        }
    }

    public function listarTarefasPorUsuario(int $usu_id): array {
        try {
            // Prepara a query para buscar os campos desejados da tabela TAREFAS
            $stmt = $this->pdo->prepare("
                SELECT 
                    TASK_ID,           
                    TASK_TITULO,       
                    TASK_TIPOTAREFA,   
                    TASK_DATAFIM,      
                    TASK_PRIORIDADE,   
                    TASK_STATUS        
                FROM TAREFAS
                WHERE USU_ID = :usu_id    
                ORDER BY TASK_DATAFIM ASC
            ");

            // Executa a query substituindo :usu_id pelo valor do usuÃ¡rio
            $stmt->execute([':usu_id' => $usu_id]);

            // Retorna todas as linhas resultantes como array associativo
            return $stmt->fetchAll();

        } catch (PDOException $e) {
            // Em caso de erro, retorna array vazio para evitar quebra da aplicaÃ§Ã£o
            return [];
        }
    }

    /**
     * Busca uma tarefa pelo ID e pelo ID do usuÃ¡rio.
     * Garante que o usuÃ¡rio sÃ³ acesse suas prÃ³prias tarefas.
     *
     * @param int $taskId ID da tarefa
     * @param int $usuarioId ID do usuÃ¡rio logado
     * @return array|null Retorna os dados da tarefa ou null se nÃ£o encontrar
     */
   public function buscarTarefaPorIdEUsuario($taskId, $usuarioId) {
    try {
        $stmt = $this->pdo->prepare("
            SELECT * FROM TAREFAS 
            WHERE TASK_ID = :taskId AND USU_ID = :usuarioId
            LIMIT 1
        ");
        $stmt->execute([
            ':taskId' => $taskId,
            ':usuarioId' => $usuarioId
        ]);

        // Busca o resultado como array associativo para facilitar o uso na view
        $resultado = $stmt->fetch(PDO::FETCH_ASSOC);

        return $resultado ?: null;

    } catch (PDOException $e) {
        // Aqui pode registrar o erro num log se desejar
        return null;
    }
}

        $stmt->execute([

function toggleMenu() {
  const menu = document.querySelector(".menu-lateral-slide");
  const estaRetraido = menu.classList.contains("retraido");

  if (estaRetraido) {
    menu.classList.add("expandindo");
    menu.classList.remove("retraido");

    menu.addEventListener("transitionend", function aoTerminarExpandir(e) {
      if (e.propertyName === "width") {
        menu.removeEventListener("transitionend", aoTerminarExpandir);
        const spans = menu.querySelectorAll("a span");
        spans.forEach(span => { span.style.opacity = "0"; });
        menu.classList.remove("expandindo");
        requestAnimationFrame(() => {
          requestAnimationFrame(() => {
            spans.forEach(span => { span.style.opacity = ""; });
          });
        });
      }
    });
  } else {
    menu.classList.remove("expandindo");
    menu.classList.add("retraido");
  }
}

document.addEventListener("DOMContentLoaded", function() {
  const splash = document.getElementById("splash");
  if (splash) {
    splash.classList.add("active");

    const messages = [
      "Colorindo post-its",
      "Desenhando emojis",
      "Etiquetando prioridades",
      "Alfinetando os alfinetes",
      "Organizando a bagunÃ§a"
    ];

    const msgEl = document.getElementById("loading-message");
    const boxes = document.querySelectorAll(".icon-box");
    const progressBar = document.querySelector(".progress-bar");
    const loading = document.getElementById("loading-screen");
    const welcome = document.getElementById("welcome-screen");

    const TEMPO_ENTRE_ICONES = 1000;
    const TEMPO_FADE = 1500;
    const TEMPO_BEM_VINDO = 2000;

    let step = 0;
    const totalSteps = boxes.length;

    const main = document.querySelector(".main-content");
    const footer = document.querySelector(".footer");
    const menu = document.querySelector(".menu-lateral-slide");

    main.classList.add("fade-target");
    main.style.display = "none";
    footer.style.display = "none";
    menu.style.display = "none";

    function showStep() {
      if (step < totalSteps) {
        // mostra Ã­cone
        boxes[step].style.opacity = 1;

        // mostra frase correspondente (inclusive a primeira)
        msgEl.textContent = messages[step];
        msgEl.style.opacity = 1;

        // atualiza barra
        progressBar.style.width = ((step + 1) / totalSteps) * 100 + "%";

        step++;
        setTimeout(showStep, TEMPO_ENTRE_ICONES);
      } else {
        // terminou carregamento â†’ fade-out do loading
        setTimeout(() => {
          loading.style.opacity = 0;
          setTimeout(() => {
            loading.style.display = "none";

            // mostra bem-vindo com fade-in
            welcome.style.display = "flex";
            setTimeout(() => {
              welcome.style.opacity = 1;

              // depois fade-out do bem-vindo
              setTimeout(() => {
                welcome.style.opacity = 0;
                setTimeout(() => {
                  splash.style.display = "none";

                  // mostra layout com fade-in
                  main.style.display = "block";
                  footer.style.display = "block";
                  menu.style.display = "block";

                  setTimeout(() => {
                    main.classList.add("visible");
                  }, 50);
                }, TEMPO_FADE);
              }, TEMPO_BEM_VINDO);
            }, 50);
          }, TEMPO_FADE);
        }, 500);
      }
    }

    showStep();
  }
});

class authModel {
    private $conn;

    public function __construct() {
        $this->conn = new mysqli("127.0.0.1", "root", "12345", "projeto_x", 3308);
        if ($this->conn->connect_error) {
            die("Erro de conexÃ£o: " . $this->conn->connect_error);
        }
    }

    public function findByLogin(string $login) {
        $stmt = $this->conn->prepare("SELECT usu_id, usu_nome, usu_apelido, usu_idade, usu_login, usu_senha FROM usuario WHERE usu_login = ?");
        $stmt->bind_param("s", $login);
        $stmt->execute();
        $result = $stmt->get_result();
        $usuario = $result->fetch_assoc();
        $stmt->close();
        return $usuario ?: null;
    }

    public function __destruct() {
        $this->conn->close();
    }
}

    $controller = $_GET['controller'];

    'auth/login',

    'auth/validaDadosParaReset',
    'auth/novaSenha',
    'auth/processaNovaSenha',
    'auth/senhaRedefinida'
];

$rotaAtual = "$controller/$method";

    require_once 'controllers/authController.php';
    if (class_exists('authController') && method_exists('authController', 'checkLogin')) {
        $sessaoAtiva = authController::checkLogin();
        if ($sessaoAtiva === false) {
            header('Location: index.php?controller=auth&method=login&expirado=1');
            exit;
        }
    } else {
        die("Erro de autenticaÃ§Ã£o: mÃ©todo checkLogin nÃ£o encontrado.");
    }

if ($controller === 'novaTarefa') {
    $controller = 'tarefa';

// Ajuste para controller 'perfil' -> 'usuario'
if ($controller === 'perfil') {
    $controller = 'usuario';
}

$controllerFile = "controllers/{$controller}Controller.php";

if (file_exists($controllerFile)) {
    require_once $controllerFile;
    $controllerClass = $controller . "Controller";

    if (class_exists($controllerClass)) {
        $obj = new $controllerClass();

        if (method_exists($obj, $method)) {
            $resultado = $obj->$method();

            // ExceÃ§Ã£o: se controller for 'home' e mÃ©todo 'index', usa 'home.php' como nome da view
            if ($controller === 'home' && $method === 'index') {
                $viewFile = "views/home/home.php";
            } else {
                $viewFile = "views/{$controller}/{$method}.php";
            }

            if (file_exists($viewFile)) {
                if (is_array($resultado)) {
                    extract($resultado);
                    $dadosParaView = $resultado;
                }

                // Captura o conteÃºdo da view via buffer
                ob_start();
                require_once $viewFile;
                $conteudo = ob_get_clean();

                // Verifica se a requisiÃ§Ã£o veio via fetch/AJAX (ex: abertura de modal)
                $ehAjax = isset($_SERVER['HTTP_X_REQUESTED_WITH'])
                    && strtolower($_SERVER['HTTP_X_REQUESTED_WITH']) === 'xmlhttprequest';

                if ($ehAjax) {
                    // RequisiÃ§Ã£o via modal -> devolve sÃ³ o conteÃºdo puro, sem layout
                    echo $conteudo;
                } else {
                    // RequisiÃ§Ã£o normal (acesso direto pela URL) -> mantÃ©m o comportamento atual
                    if (isset($_SESSION['usuario_logado'])) {
                        require_once 'layout.php';
                    } else {
                        require_once 'layout_public.php';
                    }
                }

            } else {
                echo "Erro: Arquivo da view '$viewFile' nÃ£o encontrado.";
            }
        } else {
            echo "Erro: MÃ©todo '$method' nÃ£o encontrado em $controllerClass.";
        }
    } else {
        echo "Erro: Classe do controller '$controllerClass' nÃ£o encontrada.";
    }
} else {
    echo "Erro: Arquivo do controller '$controllerFile' nÃ£o encontrado.";
